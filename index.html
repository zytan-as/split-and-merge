<script>

  // ── State ──

  const state = {

    A: { selected: new Set(['1001','1002','1003']), surviving: '1001' },

    B: { selected: new Set(['1004','1005']), surviving: '1004' },

  };


  function openModal() {

    document.getElementById('modalOverlay').classList.remove('hidden');

    document.body.style.overflow = 'hidden';

  }

  function closeModal() {

    document.getElementById('modalOverlay').classList.add('hidden');

    document.body.style.overflow = '';

  }

  function handleOverlayClick(e) {

    if (e.target === document.getElementById('modalOverlay')) closeModal();

  }

  function dismissBanner() {

    document.getElementById('combineBanner').style.display = 'none';

  }

  function stopProp(e) { e.stopPropagation(); }


  function toggleRow(group, id, event) {

    const chk = document.getElementById('chk-' + group + '-' + id);

    chk.checked = !chk.checked;

    onRowCheck(group, id, chk.checked);

  }


  function onRowCheck(group, id, checked) {

    const s = state[group];

    if (checked) {

      s.selected.add(id);

    } else {

      s.selected.delete(id);

      // If deselecting the surviving, pick the next available

      if (s.surviving === id) {

        const next = [...s.selected][0];

        if (next) setSurviving(group, next);

      }

    }

    refreshRow(group, id);

    refreshGroupHeader(group);

    refreshFooter();

  }


  function setSurviving(group, id) {

    const s = state[group];

    const old = s.surviving;

    s.surviving = id;

    document.getElementById('sur-' + group + '-' + id).checked = true;

    refreshRow(group, id);

    if (old && old !== id) refreshRow(group, old);

  }


  function onSurvivingChange(group, id) {

    setSurviving(group, id);

  }


  function refreshRow(group, id) {

    const s = state[group];

    const row = document.getElementById('row' + group + '-' + id);

    const radio = document.getElementById('sur-' + group + '-' + id);

    const radioLabel = radio.nextElementSibling;

    const badge = radio.parentElement.querySelector('.surviving-badge');

    const hint = radio.parentElement.querySelector('.surviving-hint');


    const isSelected = s.selected.has(id);

    const isSurviving = s.surviving === id;


    row.classList.toggle('surviving', isSelected && isSurviving);

    row.classList.toggle('deselected', !isSelected);


    if (radioLabel) radioLabel.style.color = isSurviving && isSelected ? '#059669' : '#9ca3af';


    // disable radio if not selected

    radio.disabled = !isSelected;

  }


  function toggleGroupAll(group, checked) {

    const s = state[group];

    const allIds = group === 'A' ? ['1001','1002','1003'] : ['1004','1005'];

    allIds.forEach(id => {

      const chk = document.getElementById('chk-' + group + '-' + id);

      chk.checked = checked;

      if (checked) s.selected.add(id);

      else s.selected.delete(id);

      refreshRow(group, id);

    });

    // Make sure surviving is valid

    if (checked && !s.selected.has(s.surviving)) {

      setSurviving(group, [...s.selected][0]);

    } else if (!checked) {

      s.surviving = null;

    }

    refreshGroupHeader(group);

    refreshFooter();

  }


  function refreshGroupHeader(group) {

    const s = state[group];

    const allIds = group === 'A' ? ['1001','1002','1003'] : ['1004','1005'];

    const allChk = document.getElementById('group' + group + '-all');

    const count = s.selected.size;

    allChk.checked = count === allIds.length;

    allChk.indeterminate = count > 0 && count < allIds.length;

    const countEl = document.getElementById('group' + group + '-count');

    countEl.textContent = count + ' shipment' + (count !== 1 ? 's' : '');

  }


  function refreshFooter() {

    const sA = state.A.selected.size;

    const sB = state.B.selected.size;

    const total = sA + sB;

    const groups = (sA >= 2 ? 1 : 0) + (sB >= 2 ? 1 : 0);

    const survivingCount = groups;


    const sumEl = document.getElementById('footerSummary');

    const btn = document.getElementById('combineBtn');


    if (total === 0) {

      sumEl.innerHTML = 'No shipments selected';

      btn.disabled = true;

      btn.textContent = 'Combine';

      return;

    }


    sumEl.innerHTML = `<strong>${groups} group${groups !== 1 ? 's' : ''}</strong> · <strong>${total} shipment${total !== 1 ? 's' : ''}</strong> selected — will be combined into <strong>${survivingCount} surviving shipment${survivingCount !== 1 ? 's' : ''}</strong>`;

    btn.disabled = groups === 0;

    btn.textContent = `Combine ${total} shipment${total !== 1 ? 's' : ''}`;

  }


  function doConfirm() {

    closeModal();

    // Show toast

    const toast = document.getElementById('toast');

    toast.classList.remove('hidden');

    setTimeout(() => toast.classList.add('hidden'), 3500);


    // Simulate list update: mark combined rows

    const combined = {

      A: { survivors: state.A.surviving, merged: [...state.A.selected].filter(x => x !== state.A.surviving) },

      B: { survivors: state.B.surviving, merged: [...state.B.selected].filter(x => x !== state.B.surviving) },

    };


    // Fade out merged rows and update survivors

    Object.entries(combined).forEach(([group, info]) => {

      info.merged.forEach(id => {

        const rowId = '#' + id;

        // Find matching row in table

        const rows = document.querySelectorAll('tbody tr');

        rows.forEach(r => {

          if (r.querySelector('.shipment-id')?.textContent === rowId) {

            r.style.transition = 'opacity 0.5s';

            r.style.opacity = '0.2';

            setTimeout(() => r.style.display = 'none', 600);

          }

        });

      });

      // Update surviving row label

      const rows = document.querySelectorAll('tbody tr');

      rows.forEach(r => {

        const idEl = r.querySelector('.shipment-id');

        if (idEl && idEl.textContent === '#' + info.survivors) {

          const badge = document.createElement('span');

          badge.className = 'combining-badge';

          badge.textContent = '✓ Combined';

          idEl.appendChild(badge);

        }

      });

    });


    // Hide banner

    dismissBanner();

  }

</script>
